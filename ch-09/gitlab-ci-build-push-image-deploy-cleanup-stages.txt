stages:
  - build
  - push
  - image
  - deploy
  - cleanup

variables:
  # Adjust these variables to match your Generic Package Registry path
  PACKAGE_NAME: "application"
  PACKAGE_VERSION: "0.0.1"
  BINARY_FILE_NAME: "app"
  ZIP_FILE_NAME: "app.zip"
  IMAGE_TAG: "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHORT_SHA"


build_code:
  stage: build
#  image: python:3.11.2-slim-buster
  image: python:latest
  before_script:
    - apt update -y && apt install binutils zip -y
  script:
    - mkdir artifacts
    - ls -la
    - cd code
    - pip install -r requirements.txt
    - pip install pyinstaller
    - pyinstaller app.py
    - ls -la
    - cd dist
    - zip -r app app
    - ls -la
    - cp app.zip ../../artifacts/
  artifacts:
     paths:
       - artifacts
  when: manual

push_code:
  stage: push
  image: alpine
  before_script:
    - apk update && apk add curl
  script:
    - ls -la
    - 'curl --header "JOB-TOKEN: $CI_JOB_TOKEN" --upload-file artifacts/app.zip "${CI_API_V4_URL}/projects/${CI_PROJECT_ID}/packages/generic/application/0.0.1/app.zip"'
  needs: ['build_code']
  when: manual



build_image:
  stage: image
  image: docker:24.0.5
  services:
    - docker:24.0.5-dind
  before_script:
    - apk add --no-cache curl unzip
    - docker login -u $CI_DEPLOY_USER -p $CI_DEPLOY_PASSWORD $CI_REGISTRY
  script:
    - 'curl --header "PRIVATE-TOKEN: $PKG_TOKEN" --fail --location "${CI_API_V4_URL}/projects/${CI_PROJECT_ID}/packages/generic/${PACKAGE_NAME}/${PACKAGE_VERSION}/${ZIP_FILE_NAME}" --output package.zip'
    - unzip -q package.zip -d extracted_app
    - docker build -t "$IMAGE_TAG" .
    - docker push "$IMAGE_TAG"
  when: manual


deploy_to_k8s:
  stage: deploy
  image: alpine/kubectl:latest
  # Uncomment if using a local GitLab Runner inside your self-hosted network
  # tags:
  #   - self-hosted-k8s
  # before_script:
  script:
    - kubectl config get-contexts
    - kubectl config use-context networknuts/lab-connection:cluster-agent
    - kubectl config get-contexts
    - kubectl create secret docker-registry gitlab-registry-secret --docker-server=registry.gitlab.com --docker-username=$CI_DEPLOY_USER --docker-password=$CI_DEPLOY_PASSWORD
    - echo "Deploying PYTHON APP to Kubernetes"
    - sed -i "s|<YOUR_IMAGE_TAG>|$IMAGE_TAG|g" k8s/deployment.yaml
    - kubectl apply -f k8s/
    - kubectl rollout status deployment/python-app
    - kubectl get pods
    - kubectl get svc
  when: manual
