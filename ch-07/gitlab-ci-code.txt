stages:
  - lint
  - build
  - scan
  - checkyaml
  - deploy
  - cleanup

# Variables
variables:
  DOCKER_TLS_CERTDIR: "/certs"
  # Use the commit SHA to uniquely tag the image 
  # during the pipeline execution
  IMAGE_TAG: $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA

# --------------------------------------
# Checking Dockerfile for best practices
# --------------------------------------
dockerfile_lint:
  stage: lint
  image: hadolint/hadolint:latest-debian
  script:
    - hadolint Dockerfile
  rules:
    #run linter only if Dockerfile changes
    - changes:
        - Dockerfile

# ---------------------------------------
# Building Image
# ---------------------------------------

build_image:
  stage: build
  image: docker:stable
  services:
    - docker:dind
  before_script:
    - docker login -u $CI_REGISTRY_USER -p $CI_REGISTRY_PASSWORD $CI_REGISTRY
  script:
    # Build the image using the commit SHA tag
    - docker build -t $IMAGE_TAG .
    # Push to GitLab's built-in container registry 
    # so the next stage can pull it
    - docker push $IMAGE_TAG

# ----------------------------------------
# Scanning image for vulnerabilities
# ----------------------------------------

trivy_scan:
  stage: scan
  image: 
    name: public.ecr.aws/aquasecurity/trivy:latest
    entrypoint: [""] # Required for GitLab runners to execute scripts properly
  cache:
    # Caches the vulnerability database to speed up consecutive pipeline runs
    key: trivy-db-cache
    paths:
      - .trivycache/
  script:
    # 1. Print out a comprehensive, human-readable report 
    #    in the GitLab CI logs
    - trivy image --cache-dir .trivycache/ $IMAGE_TAG
    
    # 2. Re-run the scan but force the pipeline to fail 
    #    ONLY if CRITICAL issues exist
    - trivy image --cache-dir .trivycache/ --exit-code 1 --severity CRITICAL $IMAGE_TAG

# ---------------------------------------
# Check Kubernetes YAML for errors
# ---------------------------------------
yaml_scan:
  stage: checkyaml
  image: alpine:latest
  before_script:
    - apk add --no-cache curl
    - curl -L https://github.com/yannh/kubeconform/releases/latest/download/kubeconform-linux-amd64.tar.gz | tar xz -C /usr/local/bin/
  script:
    - kubeconform -kubernetes-version 1.35.9 -strict -summary -output json appconfig/

# ---------------------------------------
# Deploy Application on Cluster
# ---------------------------------------
deploy_app:
  stage: deploy
  image: debian:13-slim
  variables:
    KUBECTL_VERSION: v1.34
    DEBIAN_FRONTEND: noninteractive
  script:
    # Follows https://kubernetes.io/docs/tasks/tools/install-kubectl-linux/#install-using-native-package-management
    - apt-get update
    - apt-get install -y --no-install-recommends apt-transport-https ca-certificates curl gnupg
    - curl --fail --silent --show-error --location "https://pkgs.k8s.io/core:/stable:/${KUBECTL_VERSION}/deb/Release.key" | gpg --dearmor --output /etc/apt/keyrings/kubernetes-apt-keyring.gpg
    - chmod 644 /etc/apt/keyrings/kubernetes-apt-keyring.gpg
    - echo "deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/${KUBECTL_VERSION}/deb/ /" | tee /etc/apt/sources.list.d/kubernetes.list
    - chmod 644 /etc/apt/sources.list.d/kubernetes.list
    - apt-get update
    - apt-get install -y --no-install-recommends kubectl
    - kubectl config get-contexts
    - kubectl config use-context networknuts/lab-connection:cluster-agent
    - kubectl get pods
    - kubectl get nodes -o wide
    - echo "Deploying CARDIO application to Kubernetes"
    - ls -l $CI_PROJECT_DIR/appconfig
    - kubectl apply -f $CI_PROJECT_DIR/appconfig/
    - kubectl get pods
    - kubectl get svc
